cmake_minimum_required(VERSION 3.24)
project(pendulum_native LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(core_sources ../../lib/protocol/protocol.c ../../lib/simulation/simulation.c ../../lib/device/device.c)
add_library(pendulum_core ${core_sources})
target_include_directories(pendulum_core PUBLIC ../../lib/protocol ../../lib/simulation ../../lib/device)
target_compile_options(pendulum_core PRIVATE -Wall -Wextra -Werror)
add_executable(pendulum-sim main.c)
target_link_libraries(pendulum-sim PRIVATE pendulum_core)
enable_testing()
add_executable(core-tests ../../tests/core_test.c)
target_link_libraries(core-tests PRIVATE pendulum_core)
add_test(NAME core COMMAND core-tests)
