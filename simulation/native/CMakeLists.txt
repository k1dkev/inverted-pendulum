cmake_minimum_required(VERSION 3.24)
project(pendulum_native LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(core_sources ../../lib/protocol/protocol.c ../../lib/simulation/simulation.c ../../lib/device/device.c ../../lib/policy/policy.c ../../lib/telemetry/telemetry.c)
add_library(pendulum_core ${core_sources})
target_include_directories(pendulum_core PUBLIC ../../lib/protocol ../../lib/simulation ../../lib/device ../../lib/policy ../../lib/telemetry)
target_link_libraries(pendulum_core PUBLIC m)
target_compile_options(pendulum_core PRIVATE -Wall -Wextra -Werror)
add_executable(pendulum-sim main.c)
target_link_libraries(pendulum-sim PRIVATE pendulum_core)
enable_testing()
add_executable(core-tests ../../tests/core_test.c)
target_link_libraries(core-tests PRIVATE pendulum_core)
add_test(NAME core COMMAND core-tests)
add_executable(simulation-tests ../../tests/simulation_test.c)
target_link_libraries(simulation-tests PRIVATE pendulum_core)
add_test(NAME simulation COMMAND simulation-tests)

add_executable(policy-reference-tests ../../tests/policy_reference.cpp)
target_link_libraries(policy-reference-tests PRIVATE pendulum_core)
add_test(NAME policy-reference COMMAND policy-reference-tests)
add_executable(device-feature-tests ../../tests/device_features.c)
target_link_libraries(device-feature-tests PRIVATE pendulum_core)
add_test(NAME device-features COMMAND device-feature-tests)
